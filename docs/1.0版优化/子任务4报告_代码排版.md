**B.2 问题二：两阶段随机规划 + CVaR（lp.solve_two_stage 目标函数与约束）**

```python
def solve_two_stage(price, scenD, scenG, E_start, lam=0.0, beta=0.90):
    """两阶段随机规划（含 CVaR）。
    第一阶段：计划购电 P（各情景共用）；第二阶段：按情景决策 c,f,Q,R,E。
    目标 = Σ p·P + (1/S)ΣΣ 5p·Q + λ·CVaR_β(情景总成本)。"""
    T, S = C.T_IN_DAY, len(scenD); pi = 1.0 / S
    nP = T
    c = np.zeros(nP + S * 5 * T)          # P 段 + S×(C,F,Q,R,E) 段
    c[:nP] = price                        # 计划购电费用
    for s in range(S):                    # 各情景紧急购电期望费用
        c[vary[s]['Q']] = pi * C.EMERG_MULT * price
    # CVaR 线性化：α + (1/((1-β)S)) Σ z_s，z_s ≥ ψ_s − α, z_s ≥ 0
    if lam > 0:
        alpha_idx = nvar; z_idx = nvar + 1 + np.arange(S)
        c[alpha_idx] = lam; c[z_idx] = lam / ((1 - beta) * S)
    # ... 能量守恒 + 储能递推（对每个情景）、CVaR 不等式、边界
    res = linprog(c, A_eq=A_eq, A_ub=A_ub, b_eq=..., bounds=..., method='highs')
    return dict(P=P, Q=Q, c=Cc, f=F, E=E, obj=res.fun)
```

**B.3 问题三：滚动 MPC 主循环（main_q3._rolling）**

```python
def _rolling(price, load, pv, dates, fc3, i, E_start, K=7):
    """单日滚动：0:00 计划购电 P；6/12/18 以最新光伏预报重解调整购电 A；日终固定 A 结算。"""
    load_act, pv_act, d = load[i], pv[i], dates[i]
    G0 = data.pv_forecast_kwh(fc3, d, 0)[0]         # 0:00 光伏预报
    plan = lp.solve_day_plan(price, G0, load_act, E_start)
    P, E_plan, A = plan['P'], plan['E'].copy(), plan['P'].copy()
    for tau in (6, 12, 18):                        # 三个调整时刻
        t0 = tau * 6
        G_tau = data.pv_forecast_kwh(fc3, d, tau)[0]
        E_tau = E_plan[t0 - 1] if t0 > 0 else E_start
        rp = lp.solve_adjust(price, G_tau, load_act, E_tau, P, t0)
        A[t0:] = rp['A']; E_plan[t0:] = rp['E']    # 更新调整购电与储电轨迹
    settle = lp.solve_day_fixedP(price, pv_act, load_act, E_start, A)  # 实际结算
    return dict(P=P, A=A, Q=settle['Q'], c=settle['c'], f=settle['f'], E_end=settle['E'][-1])
```
